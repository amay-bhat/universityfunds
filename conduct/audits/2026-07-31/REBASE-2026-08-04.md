## Re-derive this table

Run before any patch is applied. It reads each work order's recorded anchor and
reports where that text actually sits now:

```bash
cd ~/Projects/dashboardProject
python3 - <<'PY'
import re, glob, os
rows = []
for f in sorted(glob.glob('conduct/audits/2026-07-31/WP*.md')) + ['conduct/audits/2026-07-31/BACKLOG.md']:
    t = open(f).read()
    for blk in re.finditer(r'^### ([AB]-\d+) · .+?$(.*?)(?=^### |\Z)', t, re.M | re.S):
        m = re.search(r'\*\*Location\*\* \| `([^`]+):(\d+)`', blk.group(2))
        c = re.search(r'\*\*Current text at `HEAD` = `00a08ec`\*\*\s*\n\s*```+text\n(.*?)\n```+',
                      blk.group(2), re.S)
        if not (m and c):
            continue
        ln = int(m.group(2))
        for l in c.group(1).split('\n'):
            mm = re.match(r'^(\d+): (.*)$', l)
            if mm and int(mm.group(1)) == ln:
                rows.append((blk.group(1), m.group(1), ln, mm.group(2)))
for fid, path, ln, anchor in rows:
    if not os.path.exists(path):
        print(f'{fid:6} {path}: FILE GONE'); continue
    lines = open(path, encoding='utf-8', errors='replace').read().split('\n')
    probe = anchor.rstrip(' …')[:60]
    cur = lines[ln - 1] if len(lines) >= ln else ''
    if probe and probe in cur:
        continue                      # still exact
    hits = [i + 1 for i, l in enumerate(lines) if probe and probe in l]
    print(f'{fid:6} {path}:{ln} -> {hits[0] if hits else "NOT FOUND"}')
PY
```
